enum _AttachAction { file, gallery, camera }

/// Campo de entrada de chat con soporte multimedia y documentos
class ChatInput extends StatefulWidget {
  final void Function(
    String text,
    List<XFile>? images,
    List<DocumentAttachment>? documents,
  )? onSendMessage;
  final Future<bool> Function(String, List<XFile>?, List<DocumentAttachment>?)?
      onSubmit;
  final String? draftId;
  final DraftService? draftService;
  final bool? supportsVision;
  final VoidCallback? onChooseVisionModel;
  final void Function(DocumentAttachment)? onDocumentsAttached;
  final int sourceRevision;
  final Future<PreparedContext> Function(String, List<DocumentAttachment>, int)?
      onPreviewContext;
  final ({
    String model,
    String? conversationId,
    int messagesVersion
  })? contextRevision;
  final bool enabled;

  /// True mientras el modelo responde: se puede escribir, pero no enviar.
  final bool busy;

  /// True si hay modelos disponibles en Ollama.
  final bool connected;
  final String? hintText;

  const ChatInput({
    super.key,
    this.onSendMessage,
    this.enabled = true,
    this.onPreviewContext,
    this.contextRevision,
    this.busy = false,
    this.connected = false,
    this.hintText,
    this.onSubmit,
    this.draftId,
    this.draftService,
    this.supportsVision,
    this.onChooseVisionModel,
    this.onDocumentsAttached,
    this.sourceRevision = 0,
  }) : assert(onSendMessage != null || onSubmit != null);

  @override
  State<ChatInput> createState() => ChatInputState();
}

class ChatInputState extends State<ChatInput> {
  DraftService get _drafts => widget.draftService ?? DraftService.instance;
  final TextEditingController _controller = TextEditingController();
  final FocusNode _focusNode = FocusNode();
  final ImagePicker _imagePicker = ImagePicker();

  /// Lecturas de disco en curso, cacheadas por ruta: antes se creaba un
  /// `Future` nuevo en cada `build` y la miniatura volvía a cargar.
  final Map<String, Future<Uint8List>> _imageReads = {};

  List<XFile> _selectedImages = [];
  final List<DocumentAttachment> _documents = [];
  int _pendingExtractions = 0;
  bool _hasText = false;
  String _lastText = '';
  bool _hadDocumentContext = false;
  bool _hadPartialHistory = false;
  PreparedContext? _preview;
  String? _previewError;
  Timer? _previewTimer;
  int _previewVersion = 0;
  int _draftVersion = 0;
  int _conversationVersion = 0;
  bool _restoringDraft = false;
  bool _submitting = false;
  bool _settingDraft = false;
  String? _draftWarning;
  Timer? _draftTimer;
  final List<DocumentAttachment> _pendingDocumentActivation = [];

  List<DocumentAttachment> get documentAttachments =>
      List.unmodifiable(_documents.where((document) => document.isReady));

  ChatDraft _snapshot() => ChatDraft(
        text: _controller.text,
        imagePaths: _selectedImages.map((image) => image.path).toList(),
        documents: _documents,
      );

  Future<void> saveDraft() async {
    final id = widget.draftId;
    if (id == null || _restoringDraft) {
      return;
    }
    await _saveSnapshot(id, _snapshot());
  }

  Future<void> _saveSnapshot(String id, ChatDraft draft) async {
    try {
      final warning = await _drafts.save(id, draft);
      if (mounted && widget.draftId == id && warning != _draftWarning) {
        setState(() => _draftWarning = warning);
      }
    } catch (error) {
      if (mounted && widget.draftId == id) {
        setState(
            () => _draftWarning = 'No se pudo guardar el borrador: $error');
      }
    }
  }

  void _draftChanged() {
    if (_settingDraft) {
      return;
    }
    _draftVersion++;
    _draftTimer?.cancel();
    if (widget.draftId != null) {
      _drafts.remember(widget.draftId!, _snapshot());
      _draftTimer = Timer(
          const Duration(milliseconds: UiConfig.draftDebounceMs), saveDraft);
    }
  }

  Future<void> _restoreDraft(String id,
      {bool activateDocuments = false}) async {
    final version = ++_draftVersion;
    _restoringDraft = true;
    try {
      final saved = await _drafts.load(id);
      if (!mounted || version != _draftVersion || widget.draftId != id) {
        return;
      }
      final documents = <DocumentAttachment>[];
      for (final document in saved.documents) {
        if (document.isReady) {
          documents.add(document);
        } else {
          try {
            documents.add(document.storageId != null
                ? await DocumentStoreService.instance.hydrate(document)
                : await DocumentService.extract(document.path,
                    displayName: document.name));
          } catch (error) {
            documents
                .add(document.copyWith(error: 'No se pudo restaurar: $error'));
          }
        }
        if (!mounted || version != _draftVersion || widget.draftId != id) {
          return;
        }
      }
      setState(() {
        _settingDraft = true;
        _controller.text = saved.text;
        _settingDraft = false;
        _selectedImages = saved.imagePaths.map(XFile.new).toList();
        _documents
          ..clear()
          ..addAll(documents);
        _hasText = saved.text.isNotEmpty;
        _draftWarning = saved.warning;
      });
      if (activateDocuments) {
        for (final document
            in documents.where((document) => document.isReady)) {
          widget.onDocumentsAttached?.call(document);
        }
      }
    } catch (error) {
      if (mounted && version == _draftVersion && widget.draftId == id) {
        setState(() => _draftWarning = 'No se pudo abrir el borrador: $error');
      }
    } finally {
      if (mounted && version == _draftVersion && widget.draftId == id) {
        setState(() => _restoringDraft = false);
        refreshContextPreview();
      }
    }
  }

  void setDraftText(String text) {
    if (_restoringDraft) {
      _showSnack('Espera a que se abra el borrador antes de cambiar su texto.');
      return;
    }
    _controller.text = text;
    _controller.selection = TextSelection.collapsed(offset: text.length);
    _focusNode.requestFocus();
  }

  void refreshContextPreview() {
    _previewTimer?.cancel();
    final version = ++_previewVersion;
    setState(() {
      _preview = null;
      _previewError = null;
    });
    final callback = widget.onPreviewContext;
    if (callback == null || widget.busy || _restoringDraft) {
      return;
    }
    _previewTimer =
        Timer(const Duration(milliseconds: ContextConfig.previewDebounceMs),
            () async {
      try {
        final preview = await callback(
            _controller.text,
            _documents.where((d) => d.isReady).toList(),
            _selectedImages.length);
        if (mounted && version == _previewVersion) {
          setState(() {
            _preview = preview;
            _hadPartialHistory = preview.historyPartial;
            _hadDocumentContext = preview.documents.text.isNotEmpty ||
                preview.documents.isPartial;
            _previewError = null;
          });
        }
      } catch (error) {
        if (mounted && version == _previewVersion) {
          setState(() {
            _preview = null;
            _previewError = error.toString();
          });
        }
      }
    });
  }

  static const int maxImages = 5;
  static const int maxDocuments = 5;

  @override
  void initState() {
    super.initState();
    _controller.addListener(_onTextChanged);
    if (widget.draftId != null) {
      unawaited(_restoreDraft(widget.draftId!));
    }
  }

  @override
  void didUpdateWidget(covariant ChatInput oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (widget.draftId != oldWidget.draftId) {
      final initialDraft = oldWidget.draftId == null ? _snapshot() : null;
      if (initialDraft != null &&
          !initialDraft.isEmpty &&
          widget.draftId != null) {
        _drafts.remember(widget.draftId!, initialDraft);
      }
      _conversationVersion++;
      _draftTimer?.cancel();
      if (oldWidget.draftId != null && !_restoringDraft) {
        unawaited(_saveSnapshot(oldWidget.draftId!, _snapshot()));
      }
      _draftVersion++;
      _settingDraft = true;
      _controller.clear();
      _settingDraft = false;
      _selectedImages = [];
      _documents.clear();
      _hasText = false;
      _hadDocumentContext = false;
      _hadPartialHistory = false;
      _imageReads.clear();
      _pendingExtractions = 0;
      _pendingDocumentActivation.clear();
      _draftWarning = null;
      _restoringDraft = false;
      if (widget.draftId != null) {
        unawaited(_restoreDraft(widget.draftId!,
            activateDocuments: initialDraft?.documents.isNotEmpty == true));
      }
    }
    if (oldWidget.busy &&
        !widget.busy &&
        _pendingDocumentActivation.isNotEmpty) {
      final documents = List<DocumentAttachment>.of(_pendingDocumentActivation);
      _pendingDocumentActivation.clear();
      final conversationVersion = _conversationVersion;
      WidgetsBinding.instance.addPostFrameCallback((_) {
        if (mounted && conversationVersion == _conversationVersion) {
          for (final document in documents) {
            widget.onDocumentsAttached?.call(document);
          }
        }
      });
    }
    if (widget.busy != oldWidget.busy ||
        widget.connected != oldWidget.connected ||
        widget.draftId != oldWidget.draftId ||
        (!widget.busy &&
            (widget.contextRevision != oldWidget.contextRevision ||
                widget.sourceRevision != oldWidget.sourceRevision))) {
      refreshContextPreview();
    }
  }

  @override
  void dispose() {
    _draftTimer?.cancel();
    if (widget.draftId != null && !_restoringDraft) {
      unawaited(_saveSnapshot(widget.draftId!, _snapshot()));
    }
    _draftVersion++;
    _controller.removeListener(_onTextChanged);
    _controller.dispose();
    _focusNode.dispose();
    _imageReads.clear();
    _previewTimer?.cancel();
    _previewVersion++;
    super.dispose();
  }

  void _onTextChanged() {
    final text = _controller.text;
    if (text == _lastText) {
      return;
    }
    _lastText = text;
    if (_settingDraft) {
      return;
    }
    _draftChanged();
    refreshContextPreview();
    // Solo importa si hay texto o no; el contador se actualiza aparte.
    final hasText = _controller.text.isNotEmpty;
    if (hasText != _hasText) {
      setState(() => _hasText = hasText);
    }
  }

  /// Añade archivos soltados o seleccionados: imágenes o documentos.
  Future<void> addFiles(List<XFile> files) async {
    if (_restoringDraft) {
      _showSnack(
          'Espera a que se abra el borrador antes de adjuntar archivos.');
      return;
    }
    if (!widget.enabled || files.isEmpty) {
      return;
    }

    final conversationId = widget.draftId;
    final conversationVersion = _conversationVersion;
    final unsupported = <String>[];
    for (final file in files) {
      if (!mounted ||
          widget.draftId != conversationId ||
          conversationVersion != _conversationVersion) {
        return;
      }
      final name = _displayName(file);
      if (DocumentService.isImagePath(name)) {
        if (_selectedImages.length >= maxImages) {
          _showSnack('Máximo $maxImages imágenes adjuntas.');
          continue;
        }
        if (_selectedImages.any((image) => image.path == file.path)) {
          continue;
        }
        setState(() {
          _selectedImages.add(file);
          refreshContextPreview();
        });
        _draftChanged();
      } else if (DocumentService.isSupportedDocument(name)) {
        if (_documents.length >= maxDocuments) {
          _showSnack('Máximo $maxDocuments documentos adjuntos.');
          continue;
        }
        if (_documents.any((document) => document.path == file.path)) {
          continue;
        }
        await _addDocument(file);
        if (mounted &&
            widget.draftId == conversationId &&
            conversationVersion == _conversationVersion) {
          final added = _documents.where(
              (document) => document.path == file.path && document.isReady);
          if (added.isNotEmpty) {
            if (widget.busy) {
              _pendingDocumentActivation.add(added.first);
            } else {
              widget.onDocumentsAttached?.call(added.first);
            }
          }
          _draftChanged();
        }
      } else {
        unsupported.add(name);
      }
    }

    if (unsupported.isNotEmpty) {
      _showSnack(
        'No se puede adjuntar: ${unsupported.join(', ')}. '
        'Formatos admitidos: ${DocumentService.supportedFormatsLabel}.',
      );
    }
  }

  String _displayName(XFile file) {
    // `XFile.name` corta solo por `\` o solo por `/` según la plataforma y
    // devolvía rutas mezcladas ("carpeta/notas.txt") como nombre visible.
    final path = file.path;
    if (path.isEmpty) {
      return file.name;
    }
    return path.split(RegExp(r'[/\\]')).last;
  }

  Future<void> _addDocument(XFile file) async {
    final conversationId = widget.draftId;
    final conversationVersion = _conversationVersion;
    final name = _displayName(file);
    final placeholder = DocumentAttachment(
      path: file.path,
      name: name,
      kind: DocumentService.kindOf(name),
      sizeBytes: 0,
      isLoading: true,
    );

    setState(() {
      _documents.add(placeholder);
      _pendingExtractions++;
    });
    _draftChanged();

    final attachment = await DocumentService.extract(
      file.path,
      displayName: name,
    );

    if (!mounted ||
        widget.draftId != conversationId ||
        conversationVersion != _conversationVersion) {
      return;
    }

    setState(() {
      final index = _documents.indexWhere(
        (document) => document.path == placeholder.path && document.isLoading,
      );
      if (index >= 0) {
        _documents[index] = attachment;
      }
      if (_pendingExtractions > 0) {
        _pendingExtractions--;
      }
    });

    refreshContextPreview();
    _draftChanged();
    if (attachment.error != null) {
      _showSnack('${attachment.name}: ${attachment.error}');
    }
  }

  void _showSnack(String message) {
    if (!mounted) {
      return;
    }
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(message),
        duration: const Duration(seconds: 3),
        behavior: SnackBarBehavior.floating,
        margin: const EdgeInsets.only(bottom: 80, left: 16, right: 16),
      ),
    );
  }

  /// Abre el diálogo del sistema para elegir documentos o imágenes.
  Future<void> _pickFiles() async {
    final conversationVersion = _conversationVersion;
    try {
      final typeGroup = XTypeGroup(
        label: 'Documentos e imágenes',
        extensions: [
          ...DocumentService.supportedExtensions,
          ...DocumentService.imageExtensions,
        ],
      );
      final files = await openFiles(acceptedTypeGroups: [typeGroup]);
      if (!mounted || conversationVersion != _conversationVersion) {
        return;
      }
      if (files.isNotEmpty) {
        await addFiles(files);
      }
    } catch (e) {
      _showSnack('No se pudo abrir el selector de archivos: $e');
    }
  }

  Future<void> _pickImage(ImageSource source) async {
    final conversationId = widget.draftId;
    final conversationVersion = _conversationVersion;
    try {
      if (source == ImageSource.gallery) {
        final List<XFile> images = await _imagePicker.pickMultiImage(
          maxWidth: 2048,
          maxHeight: 2048,
          imageQuality: 85,
        );
        if (!mounted ||
            widget.draftId != conversationId ||
            conversationVersion != _conversationVersion) {
          return;
        }

        for (final image in images) {
          if (_selectedImages.length >= maxImages) {
            break;
          }
          setState(() {
            _selectedImages.add(image);
          });
        }
      } else {
        final XFile? image = await _imagePicker.pickImage(
          source: source,
          maxWidth: 2048,
          maxHeight: 2048,
          imageQuality: 85,
        );
        if (!mounted ||
            widget.draftId != conversationId ||
            conversationVersion != _conversationVersion) {
          return;
        }

        if (image != null) {
          setState(() {
            _selectedImages.add(image);
          });
        }
      }
      _draftChanged();
      refreshContextPreview();
    } catch (e) {
      debugPrint('Error picking image: $e');
      if (mounted) {
        _showSnack('Error al seleccionar imagen: $e');
      }
    }
  }

  void _removeImage(int index) {
    setState(() {
      final removedImage = _selectedImages.removeAt(index);
      _imageReads.remove(removedImage.path);
      refreshContextPreview();
    });
    _draftChanged();
  }

  void _removeDocument(int index) {
    setState(() {
      _documents.removeAt(index);
      refreshContextPreview();
    });
    _draftChanged();
  }

  Future<Uint8List> _imageBytes(int index) {
    final image = _selectedImages[index];
    return _imageReads.putIfAbsent(image.path, image.readAsBytes);
  }

  bool get _canSend =>
      widget.enabled &&
      !_restoringDraft &&
      !_submitting &&
      widget.connected &&
      (_selectedImages.isEmpty || widget.supportsVision == true) &&
      !widget.busy &&
      _pendingExtractions == 0 &&
      (_controller.text.trim().isNotEmpty ||
          _selectedImages.isNotEmpty ||
          _documents.any((document) => document.isReady));

  Future<void> _sendMessage() async {
    if (!_canSend) {
      return;
    }

    final text = _controller.text.trim();
    final images = List<XFile>.of(_selectedImages);
    final documents = _documents.where((document) => document.isReady).toList();

    final id = widget.draftId;
    final version = _draftVersion;
    setState(() => _submitting = true);
    try {
      final submit = widget.onSubmit;
      final accepted = submit == null
          ? true
          : await submit(text, images.isEmpty ? null : images,
              documents.isEmpty ? null : documents);
      if (submit == null) {
        widget.onSendMessage?.call(text, images.isEmpty ? null : images,
            documents.isEmpty ? null : documents);
      }
      if (!accepted ||
          !mounted ||
          widget.draftId != id ||
          version != _draftVersion) {
        return;
      }
      _draftTimer?.cancel();
      _controller.clear();
      setState(() {
        _selectedImages = [];
        _imageReads.clear();
        _documents.clear();
        _hasText = false;
      });
      _focusNode.requestFocus();
      await saveDraft();
    } catch (error) {
      _showSnack('No se pudo enviar: $error');
    } finally {
      if (mounted) {
        setState(() => _submitting = false);
      }
    }
  }

  /// Enter envía; Shift+Enter sigue insertando un salto de línea.
  void _handleEnter() {
    if (_canSend) {
      _sendMessage();
      return;
    }

    final value = _controller.value;
    final selection = value.selection;
    if (!selection.isValid) {
      return;
    }
    final text = value.text.replaceRange(
      selection.start,
      selection.end,
      '\n',
    );
    _controller.value = TextEditingValue(
      text: text,
      selection: TextSelection.collapsed(offset: selection.start + 1),
    );
  }

  @override
  Widget build(BuildContext context) {
    final colorScheme = Theme.of(context).colorScheme;
    final isConnected = widget.connected;

    return Container(
      decoration: BoxDecoration(
        color: colorScheme.surface,
        border: Border(
          top: BorderSide(
            color: colorScheme.outlineVariant.withValues(alpha: 0.5),
            width: 1,
          ),
        ),
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          if (_documents.isNotEmpty) _buildDocumentPreview(),
          if (_preview?.historyPartial == true ||
              (_preview == null && _hadPartialHistory))
            Padding(
              padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 4),
              child: Text(
                  _preview == null
                      ? 'La selección del historial se comprobará antes de enviar.'
                      : 'Parte del historial guardado queda fuera de este envío.',
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                  style: TextStyle(fontSize: 12, color: colorScheme.tertiary)),
            ),
          if (_draftWarning != null)
            Padding(
              padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 4),
              child: Text(_draftWarning!,
                  style: TextStyle(fontSize: 12, color: colorScheme.tertiary)),
            ),
          if (_selectedImages.isNotEmpty && widget.supportsVision != true)
            Padding(
              padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 4),
              child: Wrap(
                  crossAxisAlignment: WrapCrossAlignment.center,
                  children: [
                    Text(widget.supportsVision == null
                        ? 'No se ha comprobado si el modelo admite imágenes.'
                        : 'Este modelo no admite imágenes.'),
                    if (widget.onChooseVisionModel != null)
                      TextButton(
                          onPressed:
                              widget.busy ? null : widget.onChooseVisionModel,
                          child: const Text('Elegir modelo compatible')),
                  ]),
            ),
          if (_documents.isEmpty && _hadDocumentContext)
            Padding(
                padding:
                    const EdgeInsets.symmetric(horizontal: 16, vertical: 4),
                child: _buildContextMeter(Theme.of(context).colorScheme)),
          if (_previewError != null)
            Padding(
                padding: const EdgeInsets.all(8),
                child: Text(_previewError!,
                    style:
                        TextStyle(color: Theme.of(context).colorScheme.error))),
          if (_selectedImages.isNotEmpty) _buildImagePreview(),
          Padding(
            padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 8),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                Row(
                  crossAxisAlignment: CrossAxisAlignment.end,
                  children: [
                    _buildAttachButton(colorScheme, isConnected),
                    const SizedBox(width: 8),
                    Expanded(child: _buildTextField(colorScheme, isConnected)),
                    const SizedBox(width: 8),
                    _buildSendButton(colorScheme, isConnected),
                  ],
                ),
                const SizedBox(height: 6),
                _buildStatusLine(colorScheme),
              ],
            ),
          ),
        ],
      ),
    );
  }

  Widget _buildStatusLine(ColorScheme colorScheme) {
    final hasAttachments = _documents.isNotEmpty || _selectedImages.isNotEmpty;

    return ValueListenableBuilder<TextEditingValue>(
      valueListenable: _controller,
      builder: (context, value, _) {
        return Wrap(
          alignment: WrapAlignment.center,
          crossAxisAlignment: WrapCrossAlignment.center,
          spacing: 12,
          runSpacing: 2,
          children: [
            if (value.text.length >= 800)
              Text(
                '${value.text.length} caracteres',
                style: TextStyle(
                  fontSize: 12,
                  color: colorScheme.onSurfaceVariant,
                ),
              ),
            if (_documents.isNotEmpty)
              Text(
                '${_documents.length}/$maxDocuments documentos',
                style: TextStyle(
                  fontSize: 12,
                  color: colorScheme.onSurfaceVariant,
                ),
              ),
            if (_selectedImages.isNotEmpty)
              Text(
                '${_selectedImages.length}/$maxImages imágenes',
                style: TextStyle(
                  fontSize: 12,
                  color: colorScheme.onSurfaceVariant,
                ),
              ),
            if (!hasAttachments)
              Text(
                'Arrastra o adjunta PDF, Word, Excel o imágenes · Enter envía',
                style: TextStyle(
                  fontSize: 12,
                  color: colorScheme.onSurfaceVariant.withValues(alpha: 0.8),
                ),
              ),
          ],
        );
      },
    );
  }

  Widget _buildDocumentPreview() {
    final colorScheme = Theme.of(context).colorScheme;

    return Container(
      width: double.infinity,
      padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 8),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          Wrap(
            spacing: 8,
            runSpacing: 8,
            children: [
              for (var index = 0; index < _documents.length; index++)
                _buildDocumentChip(colorScheme, _documents[index], index),
            ],
          ),
          const SizedBox(height: 8),
          _buildContextMeter(colorScheme),
        ],
      ),
    );
  }

  /// Barra con el presupuesto de contexto que ocupan los documentos.
  Widget _buildContextMeter(ColorScheme colorScheme) {
    final preview = _preview;
    final used = preview?.documents.chars ?? 0;
    final maxChars = preview?.documentBudgetChars ?? 0;
    final ratio = maxChars <= 0 ? 0.0 : (used / maxChars).clamp(0.0, 1.0);
    final isPartial = preview?.documents.isPartial == true;
    final color = isPartial
        ? colorScheme.error
        : ratio >= 0.75
            ? colorScheme.tertiary
            : colorScheme.primary;
    // Sin plan real no se muestran límites ni recortes del cálculo legacy.
    final String status;
    if (preview == null) {
      status = _pendingExtractions > 0
          ? 'Leyendo documentos…'
          : _previewError != null
              ? 'No se pudo calcular el contexto de documentos.'
              : !widget.connected
                  ? 'Conecta Ollama para calcular el contexto de documentos.'
                  : widget.onPreviewContext == null
                      ? 'Vista previa del contexto no disponible.'
                      : widget.busy
                          ? 'El contexto se actualizará al terminar la respuesta.'
                          : 'Calculando contexto de documentos…';
    } else {
      status = isPartial
          ? 'Contexto parcial: se seleccionan fragmentos del texto'
          : 'Contexto de documentos: ${_formatChars(used)} de '
              '${_formatChars(maxChars)} caracteres'
              ' · ≈${preview.estimatedPromptTokens}/${preview.profile.tokens} tokens';
    }

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Row(
          children: [
            Icon(Icons.data_usage, size: 14, color: color),
            const SizedBox(width: 6),
            Expanded(
              child: Tooltip(
                message: status,
                child: Text(
                  status,
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                  style: TextStyle(
                    fontSize: 12,
                    color: isPartial
                        ? colorScheme.error
                        : colorScheme.onSurfaceVariant,
                  ),
                ),
              ),
            ),
          ],
        ),
        const SizedBox(height: 4),
        SizedBox(
          height: 4,
          child: preview == null
              ? null
              : ClipRRect(
                  borderRadius: BorderRadius.circular(2),
                  child: LinearProgressIndicator(
                    value: ratio,
                    minHeight: 4,
                    color: color,
                    backgroundColor: colorScheme.surfaceContainerHighest,
                  ),
                ),
        ),
      ],
    );
  }

  String _formatChars(int value) {
    if (value < 1000) {
      return '$value';
    }
    final thousands = (value / 1000).toStringAsFixed(1);
    return '${thousands.replaceAll('.0', '')} mil';
  }

  Widget _buildDocumentChip(
    ColorScheme colorScheme,
    DocumentAttachment document,
    int index,
  ) {
    final isError = document.error != null;
    final accentColor = isError ? colorScheme.error : colorScheme.primary;

    return Tooltip(
      message: document.isLoading
          ? 'Extrayendo el texto…'
          : 'Ver el texto extraído/conservado',
      child: InkWell(
        borderRadius: BorderRadius.circular(12),
        onTap: document.isLoading
            ? null
            : () => showDocumentTextDialog(context, document),
        child: Container(
          constraints: const BoxConstraints(maxWidth: 320),
          padding: const EdgeInsets.fromLTRB(10, 8, 4, 8),
          decoration: BoxDecoration(
            color: isError
                ? colorScheme.errorContainer.withValues(alpha: 0.5)
                : colorScheme.surfaceContainerHighest,
            borderRadius: BorderRadius.circular(12),
            border: Border.all(
              color: isError
                  ? colorScheme.error.withValues(alpha: 0.5)
                  : colorScheme.outlineVariant,
            ),
          ),
          child: Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              if (document.isLoading)
                SizedBox(
                  width: 18,
                  height: 18,
                  child: CircularProgressIndicator(
                    strokeWidth: 2,
                    color: accentColor,
                  ),
                )
              else
                Icon(
                  _documentIcon(document.kind),
                  size: 20,
                  color: accentColor,
                ),
              const SizedBox(width: 8),
              Flexible(
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    Text(
                      document.name,
                      maxLines: 1,
                      overflow: TextOverflow.ellipsis,
                      style: TextStyle(
                        fontSize: 13,
                        fontWeight: FontWeight.w500,
                        color: isError
                            ? colorScheme.onErrorContainer
                            : colorScheme.onSurface,
                      ),
                    ),
                    const SizedBox(height: 2),
                    Text(
                      _documentSubtitle(document),
                      maxLines: 2,
                      overflow: TextOverflow.ellipsis,
                      style: TextStyle(
                        fontSize: 11,
                        color: isError
                            ? colorScheme.onErrorContainer
                            : colorScheme.onSurfaceVariant,
                      ),
                    ),
                  ],
                ),
              ),
              IconButton(
                tooltip: 'Quitar documento',
                onPressed: () => _removeDocument(index),
                icon: const Icon(Icons.close, size: 16),
                visualDensity: VisualDensity.compact,
                color: colorScheme.onSurfaceVariant,
              ),
            ],
          ),
        ),
      ),
    );
  }

  String _documentSubtitle(DocumentAttachment document) {
    if (document.isLoading) {
      return 'Leyendo documento…';
    }
    if (document.error != null) {
      return document.error!;
    }

    final parts = <String>[
      if (document.sizeBytes > 0) document.sizeFormatted,
      '${document.extractedText?.length ?? document.charCount} caracteres conservados',
      if (document.extractionTruncated) 'extracción parcial',
    ];
    return parts.join(' · ');
  }

  IconData _documentIcon(DocumentKind kind) {
    return switch (kind) {
      DocumentKind.pdf => Icons.picture_as_pdf,
      DocumentKind.spreadsheet => Icons.table_chart,
      DocumentKind.presentation => Icons.slideshow,
      DocumentKind.text => Icons.description,
    };
  }

  Widget _buildImagePreview() {
    final colorScheme = Theme.of(context).colorScheme;
    return Container(
      height: 96,
      padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 8),
      child: ListView.builder(
        scrollDirection: Axis.horizontal,
        itemCount: _selectedImages.length,
        itemBuilder: (context, index) {
          return Padding(
            padding: const EdgeInsets.only(right: 8),
            child: Stack(
              children: [
                ClipRRect(
                  borderRadius: BorderRadius.circular(8),
                  child: FutureBuilder<Uint8List>(
                    future: _imageBytes(index),
                    builder: (context, snapshot) {
                      if (snapshot.hasData) {
                        return Image.memory(
                          snapshot.data!,
                          width: 80,
                          height: 80,
                          fit: BoxFit.cover,
                          cacheWidth: 160,
                        );
                      }
                      if (snapshot.hasError) {
                        return Container(
                          width: 80,
                          height: 80,
                          color: colorScheme.surfaceContainerHighest,
                          child: Icon(
                            Icons.broken_image_outlined,
                            color: colorScheme.onSurfaceVariant,
                          ),
                        );
                      }
                      return Container(
                        width: 80,
                        height: 80,
                        color: colorScheme.surfaceContainerHighest,
                        child: const Center(
                          child: SizedBox(
                            width: 20,
                            height: 20,
                            child: CircularProgressIndicator(strokeWidth: 2),
                          ),
                        ),
                      );
                    },
                  ),
                ),
                Positioned(
                  top: -4,
                  right: -4,
                  child: GestureDetector(
                    onTap: () => _removeImage(index),
                    child: Container(
                      width: 24,
                      height: 24,
                      decoration: BoxDecoration(
                        color: Theme.of(context).colorScheme.error,
                        shape: BoxShape.circle,
                      ),
                      child: const Icon(
                        Icons.close,
                        size: 16,
                        color: Colors.white,
                      ),
                    ),
                  ),
                ),
              ],
            ),
          );
        },
      ),
    );
  }

  Widget _buildAttachButton(ColorScheme colorScheme, bool isConnected) {
    final canAttach = widget.enabled && !_restoringDraft;

    return PopupMenuButton<_AttachAction>(
      enabled: canAttach,
      icon: const Icon(Icons.attach_file),
      tooltip: canAttach
          ? 'Adjuntar archivo, imagen o documento'
          : 'Preparando borrador…',
      onSelected: (action) {
        switch (action) {
          case _AttachAction.file:
            _pickFiles();
            break;
          case _AttachAction.gallery:
            _pickImage(ImageSource.gallery);
            break;
          case _AttachAction.camera:
            _pickImage(ImageSource.camera);
            break;
        }
      },
      itemBuilder: (context) => [
        PopupMenuItem(
          value: _AttachAction.file,
          child: Row(
            children: [
              Icon(Icons.upload_file, size: 20, color: colorScheme.primary),
              const SizedBox(width: 12),
              const Flexible(
                child: Text('Archivo (PDF, Word, Excel, PPT…)'),
              ),
            ],
          ),
        ),
        PopupMenuItem(
          value: _AttachAction.gallery,
          child: Row(
            children: [
              Icon(Icons.photo_library, size: 20, color: colorScheme.primary),
              const SizedBox(width: 12),
              const Text('Galería de imágenes'),
            ],
          ),
        ),
        if (Platform.isAndroid || Platform.isIOS)
          PopupMenuItem(
            value: _AttachAction.camera,
            child: Row(
              children: [
                Icon(Icons.camera_alt, size: 20, color: colorScheme.primary),
                const SizedBox(width: 12),
                const Text('Cámara'),
              ],
            ),
          ),
      ],
    );
  }

  Widget _buildTextField(ColorScheme colorScheme, bool isConnected) {
    return CallbackShortcuts(
      bindings: <ShortcutActivator, VoidCallback>{
        const SingleActivator(LogicalKeyboardKey.enter): _handleEnter,
        const SingleActivator(LogicalKeyboardKey.numpadEnter): _handleEnter,
        const SingleActivator(LogicalKeyboardKey.enter, control: true):
            _sendMessage,
        const SingleActivator(LogicalKeyboardKey.enter, meta: true):
            _sendMessage,
      },
      child: TextField(
        controller: _controller,
        focusNode: _focusNode,
        enabled: widget.enabled && !_restoringDraft,
        maxLines: 5,
        minLines: 1,
        textInputAction: TextInputAction.newline,
        decoration: InputDecoration(
          hintText: widget.hintText ??
              (!isConnected
                  ? 'Prepara tu mensaje; conecta Ollama para enviarlo…'
                  : widget.busy
                      ? 'Generando… puedes escribir el siguiente mensaje'
                      : 'Escribe un mensaje…'),
          hintStyle: TextStyle(color: colorScheme.onSurfaceVariant),
          border: OutlineInputBorder(
            borderRadius: BorderRadius.circular(24),
            borderSide: BorderSide.none,
          ),
          filled: true,
          fillColor: colorScheme.surfaceContainerHighest,
          contentPadding: const EdgeInsets.symmetric(
            horizontal: 20,
            vertical: 12,
          ),
        ),
      ),
    );
  }

  Widget _buildSendButton(ColorScheme colorScheme, bool isConnected) {
    final canSend = _canSend && isConnected;

    return IconButton.filled(
      onPressed: canSend ? _sendMessage : null,
      tooltip: canSend
          ? 'Enviar mensaje (Enter)'
          : !isConnected
              ? 'Conecta Ollama para enviar'
              : widget.busy || _submitting
                  ? 'Espera o detén la respuesta actual'
                  : _pendingExtractions > 0
                      ? 'Leyendo documentos…'
                      : _selectedImages.isNotEmpty &&
                              widget.supportsVision != true
                          ? 'Elige un modelo que admita imágenes'
                          : 'Escribe un mensaje o adjunta un archivo',
      icon: const Icon(Icons.send_rounded),
    );
  }
